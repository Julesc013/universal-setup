cmake_minimum_required(VERSION 3.21)
project(usk_sdk_external_consumer LANGUAGES C CXX)

set(USK_CONSUMPTION_MODE "INSTALLED" CACHE STRING "SOURCE or INSTALLED")
set(USK_LINKAGE "STATIC" CACHE STRING "STATIC or SHARED")
set(USK_FIND_VERSION "1.0.0" CACHE STRING "Requested package version")
set(USK_PACKAGE_FIXTURE "" CACHE FILEPATH "Neutral product-package fixture")
set(USK_RECIPE_FIXTURE "" CACHE FILEPATH "Neutral setup-recipe fixture")
option(USK_FIND_EXACT "Require the exact package version" ON)
option(USK_REQUIRE_UNEXPORTED_TARGET "Exercise the unexported-target refusal" OFF)

if(USK_CONSUMPTION_MODE STREQUAL "SOURCE")
  if(NOT DEFINED USK_SOURCE_DIR)
    message(FATAL_ERROR "USK_SOURCE_DIR is required in SOURCE mode")
  endif()
  add_subdirectory("${USK_SOURCE_DIR}" usk-provider EXCLUDE_FROM_ALL)
  set(UniversalSetup_CONTRACTS_DIR "${USK_SOURCE_DIR}/contracts/schema")
  set(UniversalSetup_ABI_MANIFEST "${USK_SOURCE_DIR}/contracts/abi/usk_c_abi.v1.toml")
elseif(USK_CONSUMPTION_MODE STREQUAL "INSTALLED")
  if(USK_FIND_EXACT)
    find_package(UniversalSetup ${USK_FIND_VERSION} EXACT CONFIG REQUIRED)
  else()
    find_package(UniversalSetup ${USK_FIND_VERSION} CONFIG REQUIRED)
  endif()
else()
  message(FATAL_ERROR "USK_CONSUMPTION_MODE must be SOURCE or INSTALLED")
endif()

if(USK_REQUIRE_UNEXPORTED_TARGET)
  if(NOT TARGET UniversalSetup::Lifecycle)
    message(FATAL_ERROR "UniversalSetup::Lifecycle is intentionally unavailable")
  endif()
endif()

if(NOT TARGET UniversalSetup::Headers)
  message(FATAL_ERROR "UniversalSetup::Headers is unavailable")
endif()
if(USK_LINKAGE STREQUAL "STATIC")
  set(USK_LINK_TARGET UniversalSetup::CoreStatic)
elseif(USK_LINKAGE STREQUAL "SHARED")
  set(USK_LINK_TARGET UniversalSetup::CoreShared)
else()
  message(FATAL_ERROR "USK_LINKAGE must be STATIC or SHARED")
endif()
if(NOT TARGET ${USK_LINK_TARGET})
  message(FATAL_ERROR "requested USK linkage target is unavailable: ${USK_LINK_TARGET}")
endif()

foreach(_usk_required_file IN ITEMS
    "${UniversalSetup_ABI_MANIFEST}"
    "${USK_PACKAGE_FIXTURE}"
    "${USK_RECIPE_FIXTURE}")
  if(NOT EXISTS "${_usk_required_file}")
    message(FATAL_ERROR "required SDK consumer input is unavailable: ${_usk_required_file}")
  endif()
endforeach()
unset(_usk_required_file)

function(usk_read_json_field output path)
  file(READ "${path}" _usk_json)
  string(JSON _usk_value ERROR_VARIABLE _usk_error GET "${_usk_json}" ${ARGN})
  if(NOT _usk_error STREQUAL "NOTFOUND")
    message(FATAL_ERROR "invalid JSON field in ${path}: ${_usk_error}")
  endif()
  set(${output} "${_usk_value}" PARENT_SCOPE)
endfunction()

function(usk_require_json_field path expected)
  usk_read_json_field(_usk_observed "${path}" ${ARGN})
  if(NOT _usk_observed STREQUAL "${expected}")
    message(FATAL_ERROR
      "unexpected JSON field in ${path}: expected ${expected}, observed ${_usk_observed}")
  endif()
endfunction()

set(_usk_product_schema "${UniversalSetup_CONTRACTS_DIR}/package/product_package.v1.schema.json")
set(_usk_source_schema "${UniversalSetup_CONTRACTS_DIR}/package/source_manifest.v1.schema.json")
set(_usk_recipe_schema "${UniversalSetup_CONTRACTS_DIR}/setup/product_setup_recipe.v1.schema.json")
usk_require_json_field("${_usk_product_schema}" "usk.product_package.v1" properties schema const)
usk_require_json_field("${_usk_source_schema}" "usk.source_manifest.v1" properties schema const)
usk_require_json_field("${_usk_recipe_schema}" "usk.setup_recipe.v1" properties schema const)

usk_require_json_field("${USK_PACKAGE_FIXTURE}" "usk.product_package.v1" schema)
usk_require_json_field("${USK_PACKAGE_FIXTURE}" "org.example.fixture" product_id)
usk_require_json_field("${USK_PACKAGE_FIXTURE}" "usk.source_manifest.v1" source schema)
usk_require_json_field("${USK_PACKAGE_FIXTURE}" "local_package" source kind)
usk_read_json_field(_usk_package_ref "${USK_PACKAGE_FIXTURE}" source package_ref)
if(_usk_package_ref MATCHES "^[A-Za-z][A-Za-z0-9+.-]*:" OR
   _usk_package_ref MATCHES "^[/\\\\]" OR
   _usk_package_ref MATCHES "(^|/)\\.\\.(/|$)" OR
   _usk_package_ref MATCHES "\\\\")
  message(FATAL_ERROR "neutral source reference must remain local, relative, and normalized")
endif()
usk_require_json_field("${USK_PACKAGE_FIXTURE}" "usk.component_manifest.v1" components 0 schema)
usk_require_json_field("${USK_PACKAGE_FIXTURE}" "data" mutable_paths 0)

usk_require_json_field("${USK_RECIPE_FIXTURE}" "usk.setup_recipe.v1" schema)
usk_require_json_field("${USK_RECIPE_FIXTURE}" "org.example.fixture" product_id)
usk_require_json_field("${USK_RECIPE_FIXTURE}" "required" rollback_disposition)
usk_require_json_field("${USK_RECIPE_FIXTURE}" "required" recovery_disposition)
usk_require_json_field("${USK_RECIPE_FIXTURE}" "install" lifecycle_operations 0)
usk_require_json_field("${USK_RECIPE_FIXTURE}" "uninstall" lifecycle_operations 4)
usk_require_json_field("${USK_RECIPE_FIXTURE}" "1.0.0" migrations 0 from_version)
usk_require_json_field("${USK_RECIPE_FIXTURE}" "1.1.0" migrations 0 to_version)

add_executable(usk_sdk_external_consumer main.c)
target_link_libraries(usk_sdk_external_consumer PRIVATE ${USK_LINK_TARGET})
target_compile_definitions(usk_sdk_external_consumer PRIVATE USK_SDK_CONTRACTS_VALIDATED=1)
set_property(TARGET usk_sdk_external_consumer PROPERTY C_STANDARD 99)
set_property(TARGET usk_sdk_external_consumer PROPERTY C_STANDARD_REQUIRED ON)

add_executable(usk_sdk_cpp_headers cpp_headers.cpp)
target_link_libraries(usk_sdk_cpp_headers PRIVATE UniversalSetup::Headers)
set_property(TARGET usk_sdk_cpp_headers PROPERTY CXX_STANDARD 98)
set_property(TARGET usk_sdk_cpp_headers PROPERTY CXX_STANDARD_REQUIRED ON)

if(MSVC)
  target_compile_definitions(usk_sdk_external_consumer PRIVATE _CRT_SECURE_NO_WARNINGS)
  target_compile_options(usk_sdk_external_consumer PRIVATE /W4 /WX)
  target_compile_options(usk_sdk_cpp_headers PRIVATE /W4 /WX)
else()
  target_compile_options(usk_sdk_external_consumer PRIVATE -Wall -Wextra -Wpedantic -Werror)
  target_compile_options(usk_sdk_cpp_headers PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
